Fibonacci

Find the Fibonacci number using recursive and iterative approaches.

C++
HackerRank ↗ 
#include <iostream>
using namespace std;

int fibonacci(int n)
{
    if(n <= 1)
        return n;

    return fibonacci(n - 1) + fibonacci(n - 2);
}

int main()
{
    int n;

    cout << "Enter number: ";
    cin >> n;

    cout << "Fibonacci = " << fibonacci(n);

    return 0;
}
C++
HackerRank ↗ 
#include <iostream>
using namespace std;

int main()
{
    int n;

    cout << "Enter number: ";
    cin >> n;

    int a = 0;
    int b = 1;

    if(n == 0)
    {
        cout << a;
    }
    else if(n == 1)
    {
        cout << b;
    }
    else
    {
        for(int i = 2; i <= n; i++)
        {
            int c = a + b;

            a = b;

            b = c;
        }

        cout << b;
    }

    return 0;
}

AVL Tree

AVL tree insertion with balancing and rotations.

C++
HackerRank ↗ 
node * insert(node * root, int val)
{
    if(root == NULL)
    {
        node* temp = new node;

        temp->val = val;
        temp->left = NULL;
        temp->right = NULL;
        temp->ht = 0;

        return temp;
    }

    if(val < root->val)
    {
        root->left = insert(root->left, val);
    }
    else if(val > root->val)
    {
        root->right = insert(root->right, val);
    }

    root->ht = ht_hidden(root);

    int balance = BF_hidden(root);

    if(balance > 1 && val < root->left->val)
    {
        return LL_hidden(root);
    }

    if(balance < -1 && val > root->right->val)
    {
        return RR_hidden(root);
    }

    if(balance > 1 && val > root->left->val)
    {
        return LR_hidden(root);
    }

    if(balance < -1 && val < root->right->val)
    {
        return RL_hidden(root);
    }

    return root;
}
